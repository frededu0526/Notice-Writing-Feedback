async function processAndUpload() { const pw = document.getElementById('teacherPw').value; if (pw !== 'hdh2026') return alert('교사 인증 비밀번호가 올바르지 않습니다.'); const file = pdfFileInput.files[0]; if (!file) return alert('PDF 파일을 선택해 주세요.'); const pagesPerStudent = parseInt(document.getElementById('pagesPerStudent').value, 10); const progressArea = document.getElementById('progressArea'); const progressBar = document.getElementById('progressBar'); const statusText = document.getElementById('statusText'); progressArea.classList.remove('hidden'); try { const arrayBuffer = await file.arrayBuffer(); const pdfDoc = await pdfjsLib.getDocument({ data: arrayBuffer }).promise; const totalPages = pdfDoc.numPages; const srcPdfLibDoc = await PDFLib.PDFDocument.load(arrayBuffer); let studentCount = 0; for (let startPage = 1; startPage <= totalPages; startPage += pagesPerStudent) { studentCount++; const endPage = Math.min(startPage + pagesPerStudent - 1, totalPages); statusText.textContent = `${startPage}~${endPage} 페이지 (학생 ${studentCount}) 분석 및 분할 중...`; progressBar.style.width = `${(endPage / totalPages) * 100}%`; // 1. 해당 묶음 페이지의 모든 글자 추출 let textItems = []; for (let p = startPage; p <= endPage; p++) { const page = await pdfDoc.getPage(p); const textContent = await page.getTextContent(); textContent.items.forEach(item => { if (item.str && item.str.trim()) { textItems.push(item.str.trim()); } }); } const rawText = textItems.join(' '); console.log(`[${startPage}p 원본 추출 항목]:`, textItems); // 2. "2026학년도" 같은 연도 단어 제거 let cleanText = rawText.replace(/\d{4}\s*학년도/g, ''); // 3. 만능 파싱 (줄바꿈/특수문자/공백 격파) // 3-1. 먼저 붙어있는 표준 패턴 시도 (예: 11001 홍길동) let match = cleanText.match(/(\d{4,5})\s*([가-힣]{2,4})/); let studentId = ""; let studentName = ""; if (match) { studentId = match[1]; studentName = match[2]; } else { // 3-2. 실패 시: 5자리 숫자(학번)와 2~4자 한글(이름)을 각각 분리 추출 // 학번 후보 (2026 제외 4~5자리 숫자) const idMatches = cleanText.match(/\b\d{4,5}\b/g) || []; const validIds = idMatches.filter(id => id !== "2026" && id !== "2025"); // 이름 후보 (화담고등학교, 학년도 등 무시 단어 제외 한글 2~4자) const nameMatches = cleanText.match(/[가-힣]{2,4}/g) || []; const validNames = nameMatches.filter(name => !['화담고등학교', '학년도', '공통영어', '채점결과', '성취기준', '채점기준', '배점'].includes(name) ); studentId = validIds.length > 0 ? validIds[0] : `Unknown_${studentCount}`; studentName = validNames.length > 0 ? validNames[0] : `미인식_${startPage}페이지`; } console.log(`=> 최종 결정: 학번=[${studentId}], 이름=[${studentName}]`); // 4. PDF 분할 생성 const chunkPdfDoc = await PDFLib.PDFDocument.create(); const pageIndexes = []; for (let p = startPage - 1; p < endPage; p++) { pageIndexes.push(p); } const copiedPages = await chunkPdfDoc.copyPages(srcPdfLibDoc, pageIndexes); copiedPages.forEach(p => chunkPdfDoc.addPage(p)); const pdfBytes = await chunkPdfDoc.save(); // 5. Storage 업로드 const blob = new Blob([pdfBytes], { type: 'application/pdf' }); const filePath = `${Date.now()}_${studentId}_${startPage}.pdf`; const { error: uploadError } = await db.storage .from('pdf-files') .upload(filePath, blob, { contentType: 'application/pdf' }); if (uploadError) throw uploadError; // 6. Public URL 및 DB Insert const { data: publicUrlData } = db.storage .from('pdf-files') .getPublicUrl(filePath); const { error: insertError } = await db.from(TABLE_NAME).insert({ student_id: studentId, student_name: studentName, file_url: publicUrlData.publicUrl }); if (insertError) throw insertError; } statusText.textContent = '모든 파일 업로드 완료!'; alert(`총 ${studentCount}개 항목 처리 완료! Supabase를 확인해 보세요.`); } catch (err) { console.error(err); alert('처리 중 오류 발생: ' + err.message); } }
